## 15_CHRONY

## Recommended is to use sdwdate. I always use sdwdate instead of chrony. But maintaining installation instructions for sdwdate on Arch Linux is a lot of work.
## I can make it work for Arch, but it can be broken again in a few months. If you want maximum privacy, you should take care of time synchronization with sdwdate yourself or use Kicksecure.

## Install the chrony daemon:
sudo pacman -S chronyd

## Time servers can log tracking metadata from timing requests. The servers from nothingtohide.nl claim to keep no logs.
## I should never select Cloudflare as an NTS server. Time sequences can leak metadata.
## Select a few NTS time servers you want to use. This list does not endorse any server. Vet them carefully before usage:

## to leak less metadata you should look into tirdad: https://www.kicksecure.com/wiki/Tirdad & https://github.com/0xsirus/tirdad

## Choose whether you want to use the hardened or the non-hardened chronyd. I used the Fedora chronyd services for this guide. The Arch Linux ones are less secure.
## Check the Fedora sources for a newer version https://packages.fedoraproject.org/pkgs/chrony/chrony/
## Create the hardened chronyd service. 
sudo vim /usr/lib/systemd/system/chronyd-restricted.service
```ini
# This is a more restricted version of the chronyd service intended for
# minimal NTP/NTS client configurations. The daemon is started without root
# privileges and is allowed to write only to its own runtime, state, and log
# directories. It cannot bind to privileged ports in order to operate as an
# NTP server, or provide monitoring access over IPv4/IPv6. It cannot use
# reference clocks, HW timestamping, RTC tracking, and other features.
[Unit]
Description=NTP client (restricted)
Documentation=man:chronyd(8) man:chrony.conf(5)
After=chronyd.service ntpdate.service sntp.service ntpd.service
Conflicts=chronyd.service ntpd.service systemd-timesyncd.service
ConditionCapability=CAP_SYS_TIME

[Service]
Type=notify
PIDFile=/run/chrony/chronyd.pid
Environment="OPTIONS"="-F 1"
ExecStart=/usr/sbin/chronyd -n -U $OPTIONS

User=chrony
LogsDirectory=chrony
LogsDirectoryMode=0750
RuntimeDirectory=chrony
RuntimeDirectoryMode=0750
RuntimeDirectoryPreserve=restart
StateDirectory=chrony
StateDirectoryMode=0750

AmbientCapabilities=CAP_SYS_TIME
CapabilityBoundingSet=CAP_SYS_TIME
DevicePolicy=closed
LockPersonality=yes
MemoryDenyWriteExecute=yes
NoNewPrivileges=yes
PrivateDevices=yes
PrivateTmp=yes
# This breaks adjtimex()
#PrivateUsers=yes
ProtectControlGroups=yes
ProtectHome=yes
ProtectHostname=yes
ProtectKernelLogs=yes
ProtectKernelModules=yes
ProtectKernelTunables=yes
ProtectProc=invisible
ProtectSystem=strict
RemoveIPC=yes
RestrictAddressFamilies=AF_INET AF_INET6 AF_UNIX
RestrictNamespaces=yes
RestrictRealtime=yes
RestrictSUIDSGID=yes
SystemCallArchitectures=native
SystemCallFilter=~@cpu-emulation @debug @module @mount @obsolete @raw-io
SystemCallFilter=~@reboot @resources @swap
UMask=0077

[Install]
WantedBy=multi-user.target
```

## With chrony.conf in place, start chronyd-restricted:
sudo systemctl start chronyd-restricted.service

## Check if it works without errors:
sudo systemctl status chronyd-restricted.service

## Enable it if you want to sync the time at boot:
sudo systemctl enable chronyd-restricted.service

## Add the AppArmor profile. Source: https://github.com/roddhjav/apparmor.d/blob/main/apparmor.d/profiles-a-f/chronyd
sudo vim /etc/apparmor.d/usr.sbin.chronyd
```ini
# apparmor.d - Full set of apparmor profiles
# Copyright (C) 2023-2024 Alexandre Pujol <alexandre@pujol.io>
# SPDX-License-Identifier: GPL-2.0-only

# Based on https://salsa.debian.org/debian/chrony/-/blob/debian/latest/debian/usr.sbin.chronyd

abi <abi/5.0>,

include <tunables/global>

@{exec_path} = /usr/sbin/chronyd
profile chronyd @{exec_path} flags=(attach_disconnected) {
  include <abstractions/base>
  include <abstractions/nameservice-strict>
  include <abstractions/p11-kit>
  include <abstractions/ssl_certs>

  capability chown,
  capability dac_override,
  capability dac_read_search,
  capability net_admin,
  capability net_bind_service,
  capability net_raw,
  capability setgid,
  capability setuid,
  capability sys_nice,
  capability sys_resource,
  capability sys_time,

  network inet dgram,
  network inet6 dgram,
  network inet stream,
  network inet6 stream,
  network netlink raw,

  @{exec_path} mr,

  /etc/adjtime r,
  /etc/chrony.conf r,
  /etc/chrony.keys r,
  /etc/chrony.d/{,*} r,
  /etc/chrony/{,**} r,

  /var/lib/chrony/{,*} rw,
  /var/log/chrony/{,*} rw,

  # To sign replies to MS-SNTP clients by the smbd daemon
  /var/lib/samba/ntp_signd/socket rw,

  @{run}/chrony-dhcp/{,*} r,
  @{run}/chrony.*.sock rw,
  @{run}/chrony/{,*} rw,

  # Allow reading the chronyd configuration file that timemaster(8) generates
  @{run}/timemaster/chrony.conf r,

  # Using the “tempcomp” directive gives chronyd the ability to improve
  # the stability and accuracy of the clock by compensating the temperature
  # changes measured by a sensor close to the oscillator.
  @{sys}/class/hwmon/hwmon@{int}/temp@{int}_input r,
  @{sys}/devices/virtual/thermal/thermal_zone@{int}/hwmon@{int}/temp@{int}_input r,

  /dev/pps@{int} rw,
  /dev/ptp@{int} rw,
  /dev/rtc{,@{int}} rw,

  include if exists <local/chronyd>
}

# vim:syntax=apparmor
```

## Create the AppArmor local file to give write access to systemd notify:
sudo vim /etc/apparmor.d/local/chronyd
```ini
# sd_notify() — chronyd.service is Type=notify
@{run}/systemd/notify w,
```

## Create the AppArmor drop-ins:
sudo mkdir -p /etc/systemd/system/chronyd.service.d /usr/lib/systemd/system/chronyd-restricted.service.d
sudo chmod 755 /etc/systemd/system/chronyd.service.d /usr/lib/systemd/system/chronyd-restricted.service.d
sudo vim /etc/systemd/system/chronyd.service.d/30-apparmor.conf
```ini
[Service]
## AppArmor
AppArmorProfile=chronyd
```
sudo vim /usr/lib/systemd/system/chronyd-restricted.service.d/30-apparmor.conf
```ini
[Service]
## AppArmor
AppArmorProfile=chronyd
```
sudo chmod 644 /etc/systemd/system/chronyd.service.d/30-apparmor.conf /usr/lib/systemd/system/chronyd-restricted.service.d/30-apparmor.conf /etc/apparmor.d/local/chronyd

## Enable the AppArmor profile:
sudo touch /etc/apparmor.d/local/usr.sbin.chronyd
sudo aa-enforce /etc/apparmor.d/usr.sbin.chronyd

## Reboot and check if chronyd is running and AppArmor is active:
sudo systemctl status chronyd-restricted
sudo aa-status

## After the time has synced, compare the clocks. Wait at least ~11 minutes after the sync, then:
timedatectl              # compare "Universal time" vs "RTC time"
sudo hwclock --show; date

## This chrony configuration uses iburst for initial synchronization, which allows the time servers to see when you boot the machine. To avoid boot-time correlation, you can disable the chronyd service at boot and create a cronjob that starts chronyd at a random time instead.
## If you do so, select the 1.5 second boot clock randomization in 26_VIRT_MANAGER.
